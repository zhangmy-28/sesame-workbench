    function renderContent(catKey) {
      const cat = categories[catKey];
      if (!cat) return;

       渲染头部
      const header = document.getElementById('mainHeader');
      header.innerHTML = `
        div class=category-title
          span${cat.icon}span ${cat.title}
        div
        div class=category-subtitle${cat.subtitle}div
        div class=date-badge📅 ${dateStr} 星期${dayOfWeek}div
      `;

       渲染内容
      const scroll = document.getElementById('contentScroll');
      scroll.innerHTML = '';
      scroll.scrollTop = 0;

      cat.content.forEach((item, index) = {
        const el = document.createElement('div');
        el.style.animationDelay = `${index  0.05}s`;

        if (item.type === 'dialogue-card') {
          el.className = 'dialogue-card';
          let dialogueHtml = item.dialogue.map(d =
            `div class=dialogue-linespan class=speaker${d.speaker}spanspan class=text${d.text}spandiv`
          ).join('');
          el.innerHTML = `
            div style=font-weight600;font-size14px;margin-bottom8px;colorvar(--sesame);${item.title}div
            ${dialogueHtml}
            p style=font-size11px;colorvar(--text-muted);margin-top8px;${item.tips}p
            ${item.source  `a href=${item.source.url} target=_blank rel=noopener class=source-link📎 ${item.source.label} span class=arrow→spana`  ''}
          `;
        } else if (item.type === 'daily-quote') {
          el.className = 'daily-quote';
          el.innerHTML = `
            div class=quote-text${item.text}div
            div class=quote-author${item.author}div
          `;
        } else {
           普通卡片
          el.className = 'card';
          let tagsHtml = '';
          if (item.tags) {
            tagsHtml = 'div class=card-tags' + item.tags.map(t =
              `span class=tag style=background${t.bg};color${t.color};${t.text}span`
            ).join('') + 'div';
          }
          el.innerHTML = `
            div class=card-header
              div class=card-icon style=background${item.iconBg};${item.icon}div
              div class=card-title${item.title}div
            div
            div class=card-body${item.body}div
            ${tagsHtml}
            ${item.source  `a href=${item.source.url} target=_blank rel=noopener class=source-link📎 来源：${item.source.label} span class=arrow→spana`  ''}
          `;
        }
        scroll.appendChild(el);
      });
    }

    document.getElementById('navList').addEventListener('click', (e) = {
      const navItem = e.target.closest('.nav-item');
      if (!navItem) return;

      document.querySelectorAll('.nav-item').forEach(n = n.classList.remove('active'));
      navItem.classList.add('active');

      const catKey = navItem.dataset.cat;
      renderContent(catKey);
    });
